// DCL 数据源定义表（配置层，21条固定记录）
model DCLSource {
  id          String   @id @default(uuid())
  code        String   @unique  // "DCL-01" ~ "DCL-21"
  name        String             // "经销商进销存月报表"
  sourceType  SourceType         // BI_ERP | BI_CRM | BI_SFE | TEMPLATE
  updateFreq  UpdateFreq         // MONTHLY | QUARTERLY | YEARLY
  collectMode CollectMode        // AUTO | MANUAL | HYBRID
  mclCode     String             // 关联 MCL 编号（如 "MCL-01"）
  records     DCLRecord[]
  importJobs  DCLImportJob[]
}

// DCL 采集数据记录表（数据层）
model DCLRecord {
  id          String   @id @default(uuid())
  sourceId    String
  source      DCLSource @relation(fields: [sourceId], references: [id])
  versionId   String
  version     ForecastVersion @relation(fields: [versionId], references: [id])
  buType      String?          // 业务单元，可空（全公司级数据）
  period      String           // "2026-07" 格式
  dataJson    Json             // 灵活存储各 DCL 的不同字段
  importedAt  DateTime @default(now())
  importJobId String?
  importJob   DCLImportJob? @relation(fields: [importJobId], references: [id])

  @@index([versionId, sourceId])
  @@index([versionId, buType])
}

// 导入任务追踪（每次上传 Excel 创建一条）
model DCLImportJob {
  id         String     @id @default(uuid())
  sourceId   String
  source     DCLSource  @relation(fields: [sourceId], references: [id])
  versionId  String
  fileName   String
  status     ImportStatus @default(PENDING)  // PENDING|PROCESSING|SUCCESS|FAILED
  totalRows  Int        @default(0)
  successRows Int       @default(0)
  errorMsg   String?
  createdAt  DateTime   @default(now())
  records    DCLRecord[]
}

// MCL 计算结果表（每次运行预测引擎后写入）
model MCLResult {
  id         String   @id @default(uuid())
  versionId  String
  version    ForecastVersion @relation(fields: [versionId], references: [id])
  mclCode    String           // "MCL-01" ~ "MCL-07"
  formulaKey String           // "Rev_F1" ~ "Rev_F19"
  buType     String?
  period     String           // "2026-07"
  value      Float
  calcAt     DateTime @default(now())

  @@unique([versionId, formulaKey, buType, period])
}

enum SourceType  { BI_ERP BI_CRM BI_SFE TEMPLATE }
enum UpdateFreq  { MONTHLY QUARTERLY YEARLY }
enum CollectMode { AUTO MANUAL HYBRID }
enum ImportStatus { PENDING PROCESSING SUCCESS FAILED }
✅ 设计亮点
  • dataJson: Json 字段：21个 DCL 数据结构各不相同，用 JSON 存储避免建21张表
  • MCLResult 存结果：每次计算引擎跑完写入，前端直接查结果，无需重算
  • @@unique([versionId, formulaKey, buType, period])：防止重复计算覆盖

TypeScript backend/src/services/mcl/mcl01BaselineRevenue.ts 
import { prisma } from '../../db'

/** MCL-01：基准收入引擎（Rev_F1 ~ F4） */
export class MCL01BaselineRevenue {

  /** Rev_F1 = MAX(月均出库量, 最后一月出库量) × 6 */
  async calcRevF1(versionId: string, buType: string): Promise<number> {
    const records = await prisma.dCLRecord.findMany({
      where: { versionId, buType, source: { code: 'DCL-01' } },
      orderBy: { period: 'asc' }
    })
    if (!records.length) return 0

    const outbounds = records.map(r => (r.dataJson as any).outboundQty ?? 0)
    const avgOut = outbounds.reduce((a: number, b: number) => a + b, 0) / outbounds.length
    const lastOut = outbounds[outbounds.length - 1]
    return Math.max(avgOut, lastOut) * 6
  }

  /** Rev_F2 = CRM已投放仪器数 × 单台月均产出 × 6 */
  async calcRevF2(versionId: string, buType: string): Promise<number> {
    const record = await prisma.dCLRecord.findFirst({
      where: { versionId, buType, source: { code: 'DCL-02' } }
    })
    if (!record) return 0
    const d = record.dataJson as any
    return (d.installedDevices ?? 0) * (d.monthlyOutputPerDevice ?? 0) * 6
  }

  /** Rev_F3 = 历史趋势外推 + 预计订单（线性回归外推YTD6） */
  async calcRevF3(versionId: string, buType: string): Promise<number> {
    const [ytdRec, orderRec] = await Promise.all([
      prisma.dCLRecord.findMany({ where: { versionId, buType, source: { code: 'DCL-03' } }, orderBy: { period: 'asc' } }),
      prisma.dCLRecord.findFirst({ where: { versionId, buType, source: { code: 'DCL-04' } } })
    ])
    const revenues = ytdRec.map(r => (r.dataJson as any).revenue ?? 0)
    const trend = this.linearExtrapolate(revenues, 6)  // 外推6个月之和
    const orders = (orderRec?.dataJson as any)?.totalOrderAmount ?? 0
    return trend + orders
  }

  /** Rev_F4 = 囤货月数 × 月均销售额 */
  async calcRevF4(versionId: string, buType: string): Promise<number> {
    const record = await prisma.dCLRecord.findFirst({
      where: { versionId, buType, source: { code: 'DCL-05' } }
    })
    if (!record) return 0
    const d = record.dataJson as any
    return (d.stockpileMonths ?? 0) * (d.monthlyAvgSales ?? 0)
  }

  /** MCL-01 汇总：取 MAX(F1, F2, F3) + F4（囤货计划叠加） */
  async calcMCL01Total(versionId: string, buType: string): Promise<{
    revF1: number; revF2: number; revF3: number; revF4: number; baseline: number
  }> {
    const [f1, f2, f3, f4] = await Promise.all([
      this.calcRevF1(versionId, buType), this.calcRevF2(versionId, buType),
      this.calcRevF3(versionId, buType), this.calcRevF4(versionId, buType),
    ])
    const baseline = Math.max(f1, f2, f3) + f4
    return { revF1: f1, revF2: f2, revF3: f3, revF4: f4, baseline }
  }

  /** 线性回归外推：给定历史数据点，外推未来 n 期之和 */
  private linearExtrapolate(data: number[], periods: number): number {
    if (data.length < 2) return (data[0] ?? 0) * periods
    const n = data.length
    const xs = data.map((_, i) => i + 1)
    const xMean = (n + 1) / 2
    const yMean = data.reduce((a, b) => a + b, 0) / n
    const slope = xs.reduce((acc, x, i) => acc + (x - xMean) * (data[i] - yMean), 0) /
                  xs.reduce((acc, x) => acc + (x - xMean) ** 2, 0)
    const intercept = yMean - slope * xMean
    let total = 0
    for (let i = 1; i <= periods; i++) total += intercept + slope * (n + i)
    return Math.max(total, 0)  // 不允许负收入预测
  }
}

export const mcl01 = new MCL01BaselineRevenue()
⚠️ 业务逻辑讨论

AI 初版 MCL-01 汇总用的是 F1+F2+F3+F4（简单加总），我问了财务同事后发现应该是 MAX(F1,F2,F3)+F4——三条基准线取最大值，不叠加，只有囤货计划叠加。

TypeScript backend/src/services/mcl/mcl02IncrementalRevenue.ts（节选） 
/** Rev_F9：CRM 计划投放仪器产出（按月分摊贡献） */
async calcRevF9(versionId: string, buType: string): Promise<number> {
  const records = await prisma.dCLRecord.findMany({
    where: { versionId, buType, source: { code: 'DCL-10' } }
  })
  return records.reduce((total, rec) => {
    const d = rec.dataJson as any
    const deployMonth = parseInt(rec.period.split('-')[1])   // 1-12
    const contributionMonths = Math.max(0, 12 - deployMonth + 1)
    return total + (d.plannedDevices ?? 0) * (d.monthlyOutputPerDevice ?? 0) * contributionMonths
  }, 0)
}

/** MCL-02 汇总：所有增量收入直接求和 */
async calcMCL02Total(versionId: string, buType: string) {
  const [f5, f6, f7, f8, f9] = await Promise.all([
    this.calcRevF5(versionId, buType), this.calcRevF6(versionId, buType),
    this.calcRevF7(versionId, buType), this.calcRevF8(versionId, buType),
    this.calcRevF9(versionId, buType),
  ])
  return { revF5: f5, revF6: f6, revF7: f7, revF8: f8, revF9: f9,
            incremental: f5 + f6 + f7 + f8 + f9 }
}

⏱ 40 min

费用类模型逻辑简单，但要注意"减法"语义——计算的是节约金额（负贡献），最终 EBIT = 收入 - 成本 - 净费用。

MCL-03 三条公式（Rev_F10/F11/F12）和 MCL-07 一条（Rev_F19）一起实现。注意：Rev_F12 是"降低费用"，结果应该是负数（节约）。Rev_F11 是"新增费用"，是正数（增加支出）。最终 MCL-03 净费用 = F10 + F11 - F12。

理解。我把 F12 的返回值定义为"节约金额"（正数），在 calcMCL03Net 里用减法处理，语义更清晰。F19 同理：返回"净G&A" = 基准费用 - 削减额。

/** Rev_F12 = 营销基准数 × 费用率改进目标（返回节约额，正数） */
async calcRevF12(versionId: string, buType: string): Promise<number> {
  const r = await prisma.dCLRecord.findFirst({
    where: { versionId, buType, source: { code: 'DCL-13' } }
  })
  if (!r) return 0
  const d = r.dataJson as any
  return (d.baselineSMExpense ?? 0) * (d.rateImprovement ?? 0)  // rateImprovement是小数如0.05
}

/** MCL-03 净S&M费用 = F10（维持）+ F11（新增）- F12（节约）*/
async calcMCL03Net(versionId: string, buType: string) {
  const [f10, f11, f12] = await Promise.all([
    this.calcRevF10(versionId, buType), this.calcRevF11(versionId, buType),
    this.calcRevF12(versionId, buType),
  ])
  return { revF10: f10, revF11: f11, revF12: f12, netSMExpense: f10 + f11 - f12 }
}

/** Rev_F19 = 管理部门基准费用合计 - 削减节约合计（MCL-07）*/
async calcRevF19(versionId: string): Promise<number> {
  const r = await prisma.dCLRecord.findFirst({
    where: { versionId, source: { code: 'DCL-21' } }
  })
  if (!r) return 0
  const d = r.dataJson as any
  return (d.baselineGAExpense ?? 0) - (d.cuttingAmount ?? 0)
}

⏱ 50 min

成本模型是 Plant EBIT 的核心。MCL-05 的 Rev_F17（净节约 = 效率提升 - 新增折旧）需要按月摊销 Capex。

我
实现 MCL-05 的 Rev_F17：净节约 = 人员节约和效率提升金额 - 当期新增月折旧。Capex 数据在 DCL-18，包含 totalCapex、depMonths（折旧月数）、startMonth（几月开始使用）。效率节约在 DCL-19，包含 annualSaving 和 startMonth。要按月输出 7-12月每月的净节约。
AI
好的，我按月生成7-12月数组。折旧从 startMonth 开始计，之前月份折旧为0；效率节约同理。最终每月：saving[m] = efficiency[m] - depreciation[m]。
TypeScript backend/src/services/mcl/mcl05CostReduction.ts（Rev_F17节选） 
/** Rev_F17 = 净节约（按月）= 效率节约 - 新增月折旧 */
async calcRevF17ByMonth(versionId: string): Promise<{ month: number; saving: number; depreciation: number; net: number }[]> {
  const [capexRecs, savingRecs] = await Promise.all([
    prisma.dCLRecord.findMany({ where: { versionId, source: { code: 'DCL-18' } } }),
    prisma.dCLRecord.findMany({ where: { versionId, source: { code: 'DCL-19' } } }),
  ])
  const result = []
  for (let m = 7; m <= 12; m++) {
    // 本月新增折旧 = 所有在此月或之前开始使用的Capex的月折旧之和
    const dep = capexRecs.reduce((sum, r) => {
      const d = r.dataJson as any
      if ((d.startMonth ?? 13) <= m) {
        return sum + (d.totalCapex ?? 0) / (d.depMonths ?? 60)
      }
      return sum
    }, 0)

    // 本月效率节约 = 已生效的节约项之和
    const sav = savingRecs.reduce((sum, r) => {
      const d = r.dataJson as any
      if ((d.startMonth ?? 13) <= m) {
        return sum + (d.annualSaving ?? 0) / 12
      }
      return sum
    }, 0)

    result.push({ month: m, saving: sav, depreciation: dep, net: sav - dep })
  }
  return result
}

MCL-06：研发项目预测（Rev_F18）—— 最复杂的状态机
⏱ 90 min

Rev_F18 有三条分支规则，是整个 MCL 系统中唯一需要状态机的公式。花了最多时间在这里。

📋 Rev_F18 业务规则
  • 已批准项目 → 按实际月度预算计入 FC（Firm段）
  • 计划立项项目 → 按规划预算占位计入 FC Rolling段，不计入 Firm 段
  • 未审批项目 → 仅记录在 DCL-21 管线，不计入任何费用
  • 进度偏差调整：Rev_F18 = 批准立项总金额 +/- 项目关键进度偏差调整
我
实现 Rev_F18。研发项目有三种状态：APPROVED（已批准）、PLANNED（计划立项）、PIPELINE（未审批）。APPROVED 的费用进 firmAmount，PLANNED 的进 rollingAmount，PIPELINE 的不计入。另外如果项目有进度偏差（scheduleDeviation字段，正数=超前，负数=延迟），对 firmAmount 做加减调整。
AI
明白了。scheduleDeviation 的含义是"与计划相比的月份偏差"，所以金额调整 = monthlyBudget × deviation（正数表示多完成了的月份，可以多计入；负数表示延迟，少计入）。我来实现。
我
等等，scheduleDeviation 应该是金额级别的偏差，不是月份数。直接加减就行，不用再乘 monthlyBudget。
AI
收到，已修正。偏差直接是金额，直接做 firmAmount += scheduleDeviation。
TypeScript backend/src/services/mcl/mcl06RDForecast.ts 
type RDStatus = 'APPROVED' | 'PLANNED' | 'PIPELINE'

interface RDProject {
  name: string; status: RDStatus
  approvedBudget: number       // 批准金额
  scheduleDeviation: number   // 进度偏差金额（正=超前多计，负=延迟少计）
  plannedBudget: number        // 规划预算（PLANNED 状态用）
}

export class MCL06RDForecast {

  /** Rev_F18：研发费用预测（三状态分支） */
  async calcRevF18(versionId: string): Promise<{
    firmAmount: number; rollingAmount: number; pipelineCount: number; projects: RDProject[]
  }> {
    const rec = await prisma.dCLRecord.findFirst({
      where: { versionId, source: { code: 'DCL-20' } }
    })
    const projects: RDProject[] = (rec?.dataJson as any)?.projects ?? []

    let firmAmount = 0, rollingAmount = 0, pipelineCount = 0

    for (const p of projects) {
      switch (p.status) {
        case 'APPROVED':
          // 已批准：按批准预算 + 进度偏差调整计入 Firm
          firmAmount += p.approvedBudget + (p.scheduleDeviation ?? 0)
          break
        case 'PLANNED':
          // 计划立项：仅进 Rolling 段，不计 Firm
          rollingAmount += p.plannedBudget
          break
        case 'PIPELINE':
          // 未审批：仅计数，不计入任何费用
          pipelineCount++
          break
      }
    }
    // firmAmount 不能为负（不能因为延迟调整变成负值）
    firmAmount = Math.max(firmAmount, 0)

    return { firmAmount, rollingAmount, pipelineCount, projects }
  }
}
❌ 报错 #4：业务逻辑 bug（不是代码报错）

测试时发现：一个 scheduleDeviation = -500万 的延迟项目，导致 firmAmount 计算为负数。财务规则是"延迟不能让费用变负"——最多是费用降到0。

✅ 修复

加了 firmAmount = Math.max(firmAmount, 0) 兜底，同时在单项目层面也做了保护：Math.max(p.approvedBudget + deviation, 0)（移到 switch 内）。

import ReactECharts from 'echarts-for-react'

interface MCLData {
  mcl01Baseline: number;  mcl02Incremental: number
  mcl03SMExpense: number; mcl04Cogs: number
  mcl05CostSave: number; mcl06RD: number; mcl07GA: number
}

export function MCLBreakdownPanel({ data }: { data: MCLData }) {
  const fmt = (n: number) => (n / 10000).toFixed(0) + '万'

  const items = [
    { name: 'MCL-01\n基准收入', val: data.mcl01Baseline,    sign: 1 },
    { name: 'MCL-02\n增量收入', val: data.mcl02Incremental, sign: 1 },
    { name: 'MCL-03\nS&M费用',  val: data.mcl03SMExpense,  sign: -1 },
    { name: 'MCL-04/05\n销货成本', val: data.mcl04Cogs - data.mcl05CostSave, sign: -1 },
    { name: 'MCL-06\n研发费用', val: data.mcl06RD,          sign: -1 },
    { name: 'MCL-07\n管理费用', val: data.mcl07GA,          sign: -1 },
  ]
  const ebit = data.mcl01Baseline + data.mcl02Incremental
              - data.mcl03SMExpense - (data.mcl04Cogs - data.mcl05CostSave)
              - data.mcl06RD - data.mcl07GA

  // 构建瀑布图数据
  let runningBase = 0
  const bases: number[] = [], bars: number[] = [], colors: string[] = []
  items.forEach(item => {
    const contribution = item.val * item.sign
    if (contribution >= 0) {
      bases.push(runningBase); bars.push(contribution)
      colors.push('#58a6ff')
    } else {
      bases.push(runningBase + contribution); bars.push(-contribution)
      colors.push('#ff7b72')
    }
    runningBase += contribution
  })
  // 最后一柱：EBIT
  bases.push(0); bars.push(ebit); colors.push('#3fb950')
  const labels = [...items.map(i => i.name), 'EBIT']

  return (
    <ReactECharts
      style={{ height: '320px' }}
      option={{
        tooltip: { formatter: (p: any) => `${p[1].name}: ${fmt(p[1].value)}` },
        xAxis: { type: 'category', data: labels, axisLabel: { color: '#8b949e', interval: 0 } },
        yAxis: { type: 'value', name: '万元', axisLabel: { color: '#8b949e' } },
        series: [
          { type: 'bar', stack: 'wf', data: bases, itemStyle: { color: 'transparent' } },
          { type: 'bar', stack: 'wf', data: bars.map((v, i) => ({ value: v, itemStyle: { color: colors[i] } })) }
        ]
      }}
    />
  )
}

TypeScript backend/src/services/mcl/mclOrchestrator.ts 
import { mcl01 } from './mcl01BaselineRevenue'
import { mcl02 } from './mcl02IncrementalRevenue'
import { mcl0307 } from './mcl03and07Expense'
import { mcl0405 } from './mcl04and05Cost'
import { mcl06 } from './mcl06RDForecast'
import { kpiService } from '../kpiService'

const BUS = ['DOMESTIC_MEDICAL', 'ECOMMERCE', 'INTL_PUBLIC', 'INTL_PRIVATE', 'PET']

export class MCLOrchestrator {

  async runAll(versionId: string) {
    console.log(`[MCL] 开始计算版本 ${versionId}`)

    // 对每个 BU 并行计算 MCL-01/02/03
    const buResults = await Promise.all(BUS.map(async bu => {
      const [r01, r02, r03] = await Promise.all([
        mcl01.calcMCL01Total(versionId, bu),
        mcl02.calcMCL02Total(versionId, bu),
        mcl0307.calcMCL03Net(versionId, bu),
      ])
      const revenue = r01.baseline + r02.incremental
      const smExpense = r03.netSMExpense
      return { bu, revenue, smExpense, r01, r02, r03 }
    }))

    // 公司级计算（不拆 BU）
    const [costResult, rdResult, r19] = await Promise.all([
      mcl0405.calcMCL0405(versionId),
      mcl06.calcRevF18(versionId),
      mcl0307.calcRevF19(versionId),
    ])

    // 汇总 EBIT
    const totalRevenue = buResults.reduce((s, b) => s + b.revenue, 0)
    const totalSMExpense = buResults.reduce((s, b) => s + b.smExpense, 0)
    const netCogs = costResult.netCOGS
    const rdFirm = rdResult.firmAmount
    const gaNet = r19
    const ebit = totalRevenue - netCogs - totalSMExpense - rdFirm - gaNet

    // 写入 MCLResult
    await prisma.mCLResult.upsert({
      where: { versionId_formulaKey_buType_period: { versionId, formulaKey: 'EBIT_TOTAL', buType: 'ALL', period: 'FC7-12' } },
      update: { value: ebit, calcAt: new Date() },
      create: { versionId, mclCode: 'SUMMARY', formulaKey: 'EBIT_TOTAL', buType: 'ALL', period: 'FC7-12', value: ebit }
    })

    // 触发 KPI 预警检查
    const alerts = await kpiService.batchCheckVersion(versionId)
    console.log(`[MCL] 完成，EBIT=${ebit}，触发预警 ${alerts.length} 条`)

    return { totalRevenue, netCogs, totalSMExpense, rdFirm, gaNet, ebit, alerts, buResults }
  }
}

export const mclOrchestrator = new MCLOrchestrator()

复盘：10条 MCL/DCL Vibe Coding 心法
⏱ 总计 ~6h

从思维导图到19条公式全部落地，复盘最重要的10个经验。

01
先让 AI 读文档，再编码
把思维导图 / PDF 喂给 AI 让它梳理结构，节省 70% 理解时间。不要自己先理解再解释给 AI。
02
公式是最好的 Prompt
Rev_F1 = MAX(月均出库量, 最后一月出库量) × 6 比"帮我做一个基准收入预测"强10倍。数学公式直接变代码。
03
JSON 存储异构 DCL 数据
21个 DCL 数据结构各不相同，dataJson: Json 字段比建21张表省事。扩展字段时不需要改 Schema。
04
先问难度，再定顺序
开始前问 AI "哪条公式最难"，得到 Rev_F18 > F17 > F1。按难度从简到繁，边学边迭代。
05
业务 Bug > 代码 Bug
本次6个报错中4个是业务逻辑问题（汇总方式错、费用正负号错、Capex 除零、EBIT 变负）。代码对但业务错。
06
纠正 AI 要果断、精确
"scheduleDeviation 是金额不是月数"这句纠正，AI 立刻改对了。模糊的纠正（"不对"）会让 AI 乱猜。
07
并行调用提速
MCLOrchestrator 用 Promise.all 并行跑5个 BU × 3个模型 = 15次数据库查询，从序列30s 压到 6s。
08
状态机写 switch，不写 if-else 链
Rev_F18 三分支用 switch，比 if-else 链可读性高3倍。业务规则清晰的地方 AI 也更少出错。
09
瀑布图揭示隐性问题
MCLBreakdownPanel 瀑布图上线后，财务同事第一眼就发现了 MCL-05 降本节约被重复计入的问题，靠数字看不出来。
10
兜底值比报错更友好
?? 0、Math.max(..., 0)、Math.max(..., 1) 防止 NaN/Infinity/负值传播。财务系统不能有"undefined"。
